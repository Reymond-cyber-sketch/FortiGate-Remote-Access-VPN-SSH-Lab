COMANDOS PRINCIPALES DE REFERENCIA
==================================

config system interface
edit "port2"
set alias "WEB-LAN"
set ip 10.24.96.129 255.255.255.240
set allowaccess ping
set role lan
next
end


config firewall address
edit "WEB-NET"
set subnet 10.24.96.128 255.255.255.240
next

edit "WEB-SERVER-HOST"
set subnet 10.24.96.130 255.255.255.255
next

edit "VPN-CLIENT-POOL"
set type iprange
set start-ip 10.24.97.10
set end-ip 10.24.97.20
next
end


config user local
edit "vpnuser"
set type password
set passwd <VPN_USER_PASSWORD_REDACTED>
next
end


config user group
edit "VPN-USERS"
set member "vpnuser"
next
end


config vpn ipsec phase1-interface
edit "REMOTE-SSH"
set type dynamic
set interface "port1"
set ike-version 1
set mode main
set peertype any
set net-device disable
set mode-cfg enable
set proposal des-sha256
set dhgrp 14
set dpd on-idle
set xauthtype auto
set authusrgrp "VPN-USERS"
set assign-ip-from name
set ipv4-name "VPN-CLIENT-POOL"
set ipv4-split-include "WEB-NET"
set psksecret <IPSEC_PSK_REDACTED>
next
end


config vpn ipsec phase2-interface
edit "REMOTE-SSH-P2"
set phase1name "REMOTE-SSH"
set proposal des-sha256
set pfs disable
set keepalive enable
next
end
